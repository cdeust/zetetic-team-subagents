# Python dependencies for running this repository's test suite.
#
# There is no installable package here (see pyproject.toml): the plugin's Python
# is hooks and tools that run from source. This file exists so CI and a
# contributor install the SAME set, because the coverage gate compares against a
# floor and a missing optional dependency silently changes which branches run.
#
# Everything else the tree imports is standard library, deliberately: the
# shipped surface is stdlib-only apart from the two entries marked "runtime"
# below, which keeps the install path free of a dependency tree.

# Runtime — imported by tools/acceptance_gate.py and tools/semantic_layer.py to
# read the gate configuration files. Not optional: without it, both modules fail
# at import.
PyYAML==6.0.3

# Runtime, optional by construction — tools/web_ingest.py prefers certifi's CA
# bundle and falls back to the system store inside a try/except. Pinned into the
# dev set anyway so CI exercises the SAME branch a developer does; otherwise the
# two environments measure different lines and the coverage number drifts
# between them for no reason anyone can see.
certifi==2026.7.22

# Test tooling.
# Pin every direct requirement to the version whose hashes are committed in
# requirements-dev.lock. Otherwise the CI lock-consistency check changes when
# an upstream release appears even though this repository changed nothing
# (reproduced 2026-08-02: coverage 7.15.3 superseded the locked 7.15.2).
pytest==9.1.1
coverage==7.15.2
